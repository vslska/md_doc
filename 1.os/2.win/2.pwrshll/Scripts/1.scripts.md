## Убиваем неактивных пользователей
```powershell
# 1. Получаем данные текущей сессии
$mySessionId = [System.Diagnostics.Process]::GetCurrentProcess().SessionId
$myUser      =$env:USERNAME

Write-Host "--- Жесткая чистка сервера (Мертвые + АФК сессии) ---" -ForegroundColor Cyan
Write-Host "Вы залогинены как: $myUser (ID:$mySessionId). Вы в БЕЗОПАСНОСТИ.`n" -ForegroundColor Green

# 2. Получаем данные из query user
$quser = query user 2>$null
if (!$quser) { Write-Host "Нет активных пользователей."; exit }

# Пропускаем заголовок
$quser[1..($quser.Length-1)] | ForEach-Object {
    $line = $_.Trim()
    
    # Регулярка учитывает возможный символ '>' в начале текущего пользователя
    if ($line -match '^(?<isCurrent>>)?(?<username>\S+)\s+(?<session>\S+)?\s+(?<id>\d+)\s+(?<status>Active|Disc)\s+(?<idle>\S+)') {
        
        $uName   = $Matches.username -replace '^>', ''
        $uId     = [int]$Matches.id
        $uStatus = $Matches.status
        $uIdle   = $Matches.idle
        
        # СЕБЯ НЕ ТРОГАЕМ (проверяем по маркеру '>', по ID и по имени)
        if ($Matches.isCurrent -eq '>' -or $uId -eq $mySessionId -or $uName -eq $myUser) { 
            Write-Host "Пропускаем вашу сессию: ID $uId ($uName)" -ForegroundColor Cyan
            return 
        }
        
        # Сценарий 1: Сессия отключена (Disc) — выбиваем сразу
        if ($uStatus -eq 'Disc') {
            Write-Host "Выбиваем отключенную сессию: ID $uId ($uName)" -ForegroundColor Yellow
            rwinsta $uId
            return
        }
        
        # Сценарий 2: Сессия Active, но человек ушел домой (проверяем idle time)
        if ($uStatus -eq 'Active') {
            $kickReason = $null
            
            # Если простаивает свыше суток (формат: 1+12:30)
            if ($uIdle -like '*+*') {
                $kickReason = "простой больше суток ($uIdle)"
            }
            # Если простаивает часы (формат ЧЧ:ММ, например 02:15)
            elseif ($uIdle -match '(?<hours>\d+):(?<mins>\d+)') {
                $hours = [int]$Matches.hours
                if ($hours -ge 2) { $kickReason = "простой $uIdle (больше 2 часов)" }
            }
            # Если указаны просто минуты (например, 145)
            elseif ($uIdle -match '^\d+$') {
                if ([int]$uIdle -gt 120) { $kickReason = "простой $uIdle мин (больше 2 часов)" }
            }
            
            # Если причина для кика нашлась — рубим
            if ($kickReason) {
                Write-Host "Выбиваем АФК-пользователя: ID $uId ($uName) -> Причина: $kickReason" -ForegroundColor DarkYellow
                rwinsta $uId
            }
        }
    }
}

Write-Host "`nЧистка завершена!" -ForegroundColor Green
```

## Перезапуск службы по списку (либо через файл)

```powershell
# =========================================================================
# БЛОК 1: ИНИЦИАЛИЗАЦИЯ И ПОДГОТОВКА (ПОЛНЫЙ)
# =========================================================================
Clear-Host
$StartTime = Get-Date

Write-Host "=========================================================" -ForegroundColor Cyan
Write-Host "  Перезапуск службы на серверах: $StartTime" -ForegroundColor Cyan
Write-Host "=========================================================`n" -ForegroundColor Cyan

# --- Выбор источника компьютеров ---

# ВАРИАНТ А: Ручной ввод (по умолчанию)
#$Computers = @(
#   "DNSname1"
#)

 # ВАРИАНТ Б: Чтение из файла txt (раскомментируйте при необходимости)
 $FilePath = "C:\Users\username\Documents\computers.txt"
 if (Test-Path $FilePath) {
     $Computers = Get-Content -Path $FilePath | Where-Object { -not [string]::IsNullOrWhiteSpace($_) }
 }

# --- Имя службы для перезапуска ---
$ServiceName = "SolarWindsAgent64"

# Проверка наличия целевых ПК перед стартом
if (-not $Computers -or$Computers.Count -eq 0) {
    Write-Host "[!] ОШИБКА: Список компьютеров пуст! Проверьте переменные или путь к файлу." -ForegroundColor Red
    exit
}

Write-Host "[+] Целевая служба: $ServiceName" -ForegroundColor Green
Write-Host "[+] Целевых компьютеров для обработки: $($Computers.Count)" -ForegroundColor Green


# =========================================================================
# БЛОК 2: КОД ДЛЯ ВЫПОЛНЕНИЯ НА УДАЛЕННЫХ ПК (ЛОГИКА ПЕРЕЗАПУСКА)
# =========================================================================
$ScriptBlock = {
    param($Name)
    
    try {
        # 1. Проверяем существование службы
        $Service = Get-Service -Name $Name -ErrorAction SilentlyContinue
        if (-not $Service) {
            return [PSCustomObject]@{
                Status  = "Ошибка"
                Message = "Служба '$Name' не найдена"
            }
        }

        # 2. Перезапускаем службу
        Restart-Service -Name $Name -Force -ErrorAction Stop

        # 3. Небольшая пауза
        Start-Sleep -Seconds 3

        # 4. Проверяем итоговый статус
        $FinalStatus = (Get-Service -Name $Name).Status

        return [PSCustomObject]@{
            Status  = $FinalStatus.ToString()
            Message = "Служба успешно перезапущена"
        }
    }
    catch {
        return [PSCustomObject]@{
            Status  = "Ошибка"
            Message = $_.Exception.Message
        }
    }
}


# =========================================================================
# БЛОК 3: ВЫПОЛНЕНИЕ И ОТЧЕТНОСТЬ (ПОЛНЫЙ)
# =========================================================================
Write-Host "`n[>] Отправка команд через WinRM..." -ForegroundColor Yellow

# Запуск с параллельной обработкой
$Result = Invoke-Command -ComputerName $Computers `
                         -ScriptBlock $ScriptBlock `
                         -ArgumentList $ServiceName `
                         -ThrottleLimit 10 `
                         -ErrorVariable RemotingErrors | 
          Select-Object @{N='Сервер'; E={$_.PSComputerName}}, 
                        @{N='Статус службы'; E={$_.Status}}, 
                        @{N='Детали'; E={$_.Message}}

# Вывод результатов выполнения
if ($Result) {
    Write-Host "`n--- РЕЗУЛЬТАТЫ ВЫПОЛНЕНИЯ ---" -ForegroundColor Gray
    $Result | Format-Table -AutoSize
}

# Отдельный отчет по недоступным ПК (офлайн, нет WinRM, ошибки доступа)
if ($RemotingErrors) {
    Write-Host "`n--- НЕ УДАЛОСЬ ПОДКЛЮЧИТЬСЯ К ПК ---" -ForegroundColor Red
    $FailedHosts = $RemotingErrors.TargetObject | Select-Object -Unique
    foreach ($HostName in $FailedHosts) {
        Write-Host " [X] $HostName" -ForegroundColor Red
    }
}

$ExecutionTime = (Get-Date) - $StartTime
Write-Host "`n=========================================================" -ForegroundColor Cyan
Write-Host "  Завершено за: $($ExecutionTime.Minutes) мин. $($ExecutionTime.Seconds) сек." -ForegroundColor Cyan
Write-Host "=========================================================" -ForegroundColor Cyan
```
